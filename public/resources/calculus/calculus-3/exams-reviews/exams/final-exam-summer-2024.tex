%%%%% PROBLEM 1 %%%%%
\newpage
\begin{problem}[1. (20 points)]
Find $\frac{\partial f}{\partial t}$ where $f(x,y) = e^{xy}+ x^2y$, $x = 2u-v$, $y = u^2+v^2$, $u = 3s-4t, v = \sqrt{s}$.
\end{problem}


%%%%% PROBLEM 2 %%%%%
\newpage
\begin{problem}[2. (20 points)]
Evaluate the following limit. Write ``Does not exist" if applicable.

\begin{equation*}
    \lim_{(x,y) \to (0,0)} \frac{xy}{x^2+y^2 + (x-y)^2}.
\end{equation*}
\end{problem}



%%%%% PROBLEM 3 %%%%%
\newpage
\begin{problem}[3. (20 points)]
Consider the following surface.

\begin{equation*}
    f(x,y) = 4xy-x^4-y^4.
\end{equation*}

Find \emph{and} classify \textbf{ALL} relative maximum,  minimums, and saddle points.
\end{problem}


%%%%% PROBLEM 4 %%%%%
\newpage
\begin{problem}[4. (20 point)]
Consider the following surface.

\begin{equation*}
    z = y^2\ln{x}
\end{equation*}

\vspace{.125cm}

(i) Find the slope of the tangent line to the surface at $(e,1)$ in the direction of $\vec{u} = \langle 2, 3 \rangle$.

\vspace{.25cm}

(ii) Find the equation of the tangent plane to the surface at the point $(e,1,1)$.
\end{problem}

%%%%% PROBLEM 7 %%%%%
\newpage
\begin{problem}[7. (20 points)]
Set up \textbf{BUT DO NOT EVALUATE} the following double integral using the transformation $u = \frac{y}{x}$ and $v = xy$.

\begin{equation*}
\iint_R e^{xy} \dif{A}
\end{equation*}

\vspace{.25cm}

Where $R$ is bounded by the lines $y=\frac{1}{2}x$, $y=x$, and the hyperbolas $y=\frac{1}{x}$, and $y=\frac{2}{x}$.

\end{problem}


%%%%% PROBLEM 8 %%%%%
\newpage
\begin{problem}[8. (20 points)]
Evaluate the following line integral where $C$ is the curve formed by the circle $x^2+y^2=1$ and $y = |x|$.

\begin{equation*}
    \oint_C \sqrt{x}\dif{x} +  (xy +e^{\tan{y}}) \dif{y}
\end{equation*}

\end{problem}



%%%%% PROBLEM 9 %%%%%
\newpage
\begin{problem}[9. (20 points)]
Consider the vector field $F(x,y) = (2xy^2+2y) \hat{i} + (2x^2y+2x)\hat{j}$

\vspace{.125cm}


(i) Find the divergence of $F$ at the point $(2,3)$.

\vspace{.125cm}

(ii) Find the curl of $F$ at the point $(2,3)$


\vspace{.125cm}

(iii) Find the amount of work to go through $F$ from $(1,1)$ to $(2,2)$.

\end{problem}
