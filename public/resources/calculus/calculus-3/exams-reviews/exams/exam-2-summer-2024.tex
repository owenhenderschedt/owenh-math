\newpage
\begin{problem}[1. (12 points)]
Find\emph{all} first and second order partial derivatives of the function

\begin{equation*}
    f(x,y) = x^2e^{xy}
\end{equation*}
\end{problem}


%%%%% PROBLEM 2 %%%%%
\newpage
\begin{problem}[2. (12 points)]
Find $\frac{\partial z}{\partial t}$ and $\frac{\partial z}{\partial s}$ at $s=1, t=2,u=3$, if $z = x^4+x^2y$, $x=s+2t-u$, and $y=stu^2$.
\end{problem}



%%%%% PROBLEM 3 %%%%%
\newpage
\begin{problem}[3. (12 points)]
Find \emph{all} local maximum, minimum, and saddle point values of the function

\begin{equation*}
    f(x,y)=x^2-xy+y^2+9x-6y+5.
\end{equation*}
\end{problem}


%%%%% PROBLEM 4 %%%%%
\newpage
\begin{problem}[4. (10 point)]
Find the derivative of $f(x,y)$ in the direction of the vector $u = \langle 2,-4\rangle$.

\begin{equation*}
    f(x,y) = 2x^3y-\frac{x}{y}
\end{equation*}
\end{problem}




%%%%% PROBLEM 5 %%%%%
\newpage
\begin{problem}[5. (10 points)]
Find the equation for the tangent plane and the normal line to the following surface at the point $(1,1,1)$.

\begin{equation*}
    yz + x\ln{y} = z^2.
\end{equation*}
\end{problem}


%%%%% PROBLEM 6 %%%%%
\newpage
\begin{problem}[6. (10) points)]
Evaluate the following limit. Write \emph{``Does not exist"} if needed.

\begin{equation*}
    \lim_{(x,y)\to (0,0)} \frac{xy^3+x^3y}{x^4+y^4+x^2y^2}.
\end{equation*}
\end{problem}

\newpage
\begin{problem}[7. (10 points)]
Consider the surface $f(x,y) = \sin{(2x+2y)}$. 
\begin{enumerate}
    \item Find $f_{xx}$
    \item Find $f_{xxyy}$
    \item Find $f_{xxyy \cdots xxyy}$ if there are $k$-copies of $xxyy$
\end{enumerate}
\end{problem}


%%%%% PROBLEM 8 %%%%%
\newpage
\begin{problem}[8. (12 points)]
Consider the following 4-dimensional surface.

\begin{equation*}
    f(x,y,z) = 2x^3z+3xyz + yz^2.
\end{equation*}

(a) Find the gradient $\nabla f(x,y,z)$.

\vspace{.25cm}

(b) Find the direction of steepest ascent from the point $f(1,1,-1)$.

\vspace{.25cm}

(c) Find the slope of the steepest ascent found in (b)
\end{problem}

Find the absolute maximum and minimum of the surface $f(x,y) = x^2+xy+y^2$ bounded by $R = \{(x,y) : 0\leq x\leq 1, 0\leq y\leq 2\}$.
\end{problem}
