%%%%% PROBLEM 1 %%%%%
\newpage
\begin{problem}[1. (12 points)]
Evaluate the following double integral where $R$ is the region bounded by the first quadrant, $x=3$ and $y = \sqrt{x}$.

\begin{equation*}
    \iint_R \frac{y}{x^2+1} \dif{A}
\end{equation*}
\end{problem}


%%%%% PROBLEM 2 %%%%%
\newpage
\begin{problem}[2. (10 points)]
Estimate the volume under the surface $f(x,y)$, over the region $R = \{(x,y): 0\leq x\leq 2, 0\leq y \leq 4\}$ using upper right corners, $m=2$ and $n=2$.

\begin{equation*}
    f(x,y) = \frac{xy}{x+y+1}
\end{equation*}
\end{problem}



%%%%% PROBLEM 3 %%%%%
\newpage
\begin{problem}[3. (15 points)]
Find the volume of $T$, where $T$ is the solid bounded by the $xy$-plane, $z = 3+x$, $x^2+y^2 = 4$, and $x^2+y^2 = 1$.
\end{problem}


%%%%% PROBLEM 4 %%%%%
\newpage
\begin{problem}[4. (12 point)]
Use spherical coordinates to find an expression for the volume of the region bounded by the first octant, $x^2+y^2+z^2 = 2z$ and $x^2+y^2+z^2=4$. \textbf{DO NOT EVALUATE!}
\end{problem}




%%%%% PROBLEM 5 %%%%%
\newpage
\begin{problem}[5. (12 points)]
Consider the following double integral.

\begin{equation*}
    \iint_R f(x,y) \dif{A} = \int_0^1 \int_y^{\sqrt{y}} x-y \dif{x}\dif{y}
\end{equation*}

\vspace{.25cm}

(a) Reverse the order of integration.

\vspace{.25cm}

(b) Evaluate the integral.

\vspace{.25cm}

(c) If the area of the $R$ is $\frac{1}{6}$ find the average value of $f(x,y)$.
\end{problem}


%%%%% PROBLEM 6 %%%%%
\newpage
\begin{problem}[6. (12 points)]
Prove using integration that the volume of the region between spheres $x^2+y^2+z^2=4$ and $x^2+y^2+z^2= 9$ is $\frac{76\pi}{3}$.
\end{problem}


%%%%% PROBLEM 7 %%%%%
\newpage
\begin{problem}[7. (15 points)]
Set up \textbf{BUT DO NOT EVALUATE} the following double integral using the transformation $x=\frac{1}{2}(u+v)$ and $y= \frac{1}{2}(u-v)$.

\begin{equation*}
\iint_R \frac{4}{(x-y)^2}\dif{x}\dif{y}
\end{equation*}

\vspace{.25cm}

Where $R$ is the trapezoid bounded by the lines $x-y=2$, $x-y=4$, $x=0$, and $y=0$.

\end{problem}
