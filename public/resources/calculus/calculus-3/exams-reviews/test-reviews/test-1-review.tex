\section{Introduction to Vectors}

\begin{problem}
If $\vec{u} = 5\hat{i}+3\hat{j}-2\hat{k}$ what is the magnitude of $\vec{w} = -2\vec{u}$. What is a unit vector in the same direction as $w$? Is $\langle -20, -12, 8 \rangle$ parallel to $\vec{u}$ and $\vec{w}$?  
\end{problem}

\begin{problem}
Let $\vec{a} = \langle 1,-3,-3 \rangle$ and $\vec{b} = \langle -5, 2, 1 \rangle$. Find the following:
\begin{enumerate}
    \item $2\vec{a} \cdot \vec{b}$
    \item $\vec{a} \times \vec{b}$ and $\vec{b} \times \vec{a}$.
    \item $comp_{b}(a)$ and $proj_{b}(a)$
    \item The angle between $\vec{a}$ and $\vec{b}$
    \item The area of the triangle formed by $\vec{a}$ and $\vec{b}$.
\end{enumerate}
\end{problem}

\begin{problem}
Find the direction cosines to the vector $\langle -2,-3,1\rangle$
\end{problem}


\section{Lines in 3 Dimensions}

\begin{problem}
Find the equation of a line which goes through $P(4,4,2)$ in the direction $\langle \frac{1}{2}, \frac{1}{3}, \frac{1}{4} \rangle$.
\end{problem}
\begin{problem}
Given the line $L$ written in parametric equation as $x(t) = 2t-7$, $y(t) = 12-5t$ and $z(t) = t+4$, write $L$ in symmetric form. Where will $L$ cross the plane $y=0$?
\end{problem}

\begin{problem}
Find the equation of the line through the point $P(-1,-2,4)$ which is parallel to the line $x(t) = t, y(t) = -5t+1, z(t) = 2t+2$.  
\end{problem}

\begin{problem}
    Find an equation of any line which is orthogonal to the plane $2(x-3)+6(y+2)-4(z-1) = 0$.
\end{problem}

\begin{problem}
    Will $L_1$ and $L_2$ intersect? If not, are they parallel or skew? \[L_1: x= t+1, y = t+2, z= 2t+3\] and \[L_2: x = 3t-2, y= 5t, z = 3t+1\]
\end{problem}

\section{Planes in 3 Dimensions}

\begin{problem}
    Find the equation of the plane which goes through $P(10,20,30)$ which is normal to the vector $5\hat{i}-\hat{j}-\hat{k}$.
\end{problem}

\begin{problem}
    Find the equation of the plane containing the three points $P_1(1,2,3)$, $P_2(2,4,6)$, and $P_3(-1,-3,-5)$.
\end{problem}

\begin{problem}
    Find the distance from the plane $2x-5y+z = 0$ and the point $P(-2,-7,3)$.
\end{problem}

\begin{problem}
    Find the equation of the plane which contains the point $P(4,-1,-1)$ and the line $\frac{x}{2}=\frac{y-3}{3}=z-1$
\end{problem}

\section{Introduction to Vector Functions}

\begin{problem}
What is the domain of the vector function $\langle \frac{1}{t}, \frac{1}{t+1}, \frac{1}{t+2}\rangle$?
\end{problem}

\begin{problem}
What is the domain of the vector function $\langle \frac{1}{e^t-1}, \sqrt[3]{t+5}, \ln(t+5)\rangle$
\end{problem}

\begin{problem}
    What is the domain of the vector function $\frac{1}{\sin(t)}\hat{i} + \frac{1}{\cos(t)}\hat{j} + \tan^{-1}(t)\hat{k}$
\end{problem}

\begin{problem}
    Sketch the space curve given by the vector function $\vec{r}(t) = \sin^2(t)\hat{i}+\cos^2{t}\hat{j}+t\hat{k}$
\end{problem}

\section{Calculus on Vector functions}

\begin{problem}
    Find $\lim_{t\to 2} \vec{r}(t)$ where,\[\vec{r}(t) = \left\langle \frac{e^t-e^2}{t^2-4}, \cos \left(\frac{\pi t}{2}\right), t+t^2+t^3\right\rangle\]
\end{problem}

\begin{problem}
    Find $\lim_{t\to \infty} \vec{r}(t)$ where \[\vec{r}(t) = \left\langle \frac{\sqrt{t^4-2t+7}}{3t^2-9}, e^{-\sqrt{t}}, \frac{\ln{t}}{t}\right\rangle\]
\end{problem}

\end{problem}

\end{problem}
