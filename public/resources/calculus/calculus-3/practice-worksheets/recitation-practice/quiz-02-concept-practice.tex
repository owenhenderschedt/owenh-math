\sectionbar{A. Planes and Perpendicular Geometry}

Let $P=(1,1,0)$ and $Q=(5,-1,4)$, and let $X$ be the set of
all points equidistant from $P$ and $Q$.

\begin{parts}

\item Find an equation of the plane $X$.

\item Let $M$ be the midpoint of $\overline{PQ}$, and let $\ell$ be
the line through $M$ with direction vector $\langle1,2,0\rangle$.
Find an equation of the plane $Y$ that contains $\ell$ and is
perpendicular to $X$. Also give a unit normal vector to $Y$.

\item Find the point on $X$ closest to the origin.

\end{parts}

\sectionbar{B. Triangle Geometry and Cross Product}

A triangle has vertices
\[
U=(0,0,0),\qquad V=(4,0,0),\qquad W=(1,a,2a).
\]

\begin{parts}

\item Find the area of the triangle in terms of $a$.

\item Find all values of $a$ for which the area is $6\sqrt5$.

\end{parts}

\sectionbar{C. Projections and Parallel Lines}

Consider the parallel lines
\[
L_1:\quad x=2+t,\qquad y=2+2t,\qquad z=3+t,
\]
and
\[
L_2:\quad x=5+s,\qquad y=7+2s,\qquad z=2+s.
\]
Let $A=(2,2,3)$ be the point on $L_1$ corresponding to $t=0$, and
let $B=(5,7,2)$ be the point on $L_2$ corresponding to $s=0$.

\begin{parts}

\item Find the projection of $\overrightarrow{AB}$ onto the direction
of the lines.

\item Find the distance between $L_1$ and $L_2$.

\end{parts}

\vspace{0.12em}

\sectionbar{D. Angles and Unit Vectors and Dot Product}

Let $\theta$ be the angle between vectors $\mathbf u$ and $\mathbf v$.
Suppose
\[
|\mathbf u|=3,\qquad
|\mathbf v|=5,\qquad
(\mathbf u+2\mathbf v)\cdot(3\mathbf u-\mathbf v)=27.
\]

\begin{parts}

\item Find $\cos\theta$.

\item Let $\mathbf q=\langle2,1,2\rangle$. Find all unit vectors
$\mathbf w=\langle a,0,b\rangle$ that make the same angle $\theta$
with $\mathbf q$.

\end{parts}
