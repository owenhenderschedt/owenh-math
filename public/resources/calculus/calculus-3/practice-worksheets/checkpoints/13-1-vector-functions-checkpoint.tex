\vs\vs

\begin{problem}
If $\vec{r}(t) = \langle t, e^t, e^{-t} \rangle$, what is $\vec{r}(0)$ and what does it mean?  
\end{problem}

\begin{problem}
what is $\lim_{t\to 3} \langle \frac{1}{t}+\frac{1}{t^2}+\frac{1}{t^3}\rangle$
\end{problem}

\begin{problem}
If $\vec{r'}(t) = (\sqrt{t})\hat{i}+(\sqrt{2t})\hat{j}+(\sqrt{3t})\hat{k}$, what is the corresponding unit tangent vector?
\end{problem}


\begin{problem}
Find $\vec{r'}(t)$ where $\vec{r}(t) = \langle \ln{t}, \sin{t}, \tan{t}\rangle$.
\end{problem}

\begin{problem}
Find $\int \vec{r}(t) dt$ where $\vec{r}(t) = \langle e^{3t}, t^2, \cos{t} \rangle$
\end{problem}


\newpage

\section{Answers}

\begin{enumerate}
    \item $\vec{r}(0) = \langle 0,1,1 \rangle$ it is a $3$-dimensional position vector
    \item $\langle 1/3,1/9,1/27\rangle$
    \item $T(t) = \frac{1}{\sqrt{6t}}((\sqrt{t})\hat{i}+(\sqrt{2t})\hat{j}+(\sqrt{3t})\hat{k})$
    \item $\vec{r'}(t) = \langle \frac{1}{t}, \cos{t}, \sec^2{t}\rangle $
    \item $\int \vec{r}(t) dt = \langle \frac{1}{3}e^{3t}, \frac{1}{3}t^3, \sin{t} \rangle + \langle c_1, c_2, c_3 \rangle$ or $+C$ where $C$ is a vector.
\end{enumerate}
