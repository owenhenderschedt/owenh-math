\section{Problems}

\begin{problem}
Find $\langle 2,6,-1\rangle \cdot \langle -3,0,4\rangle$.   
\end{problem}

\begin{problem}
    If $\Vec{u} = -2\hat{i}-\hat{j}+\hat{k}$ and $\Vec{v}= 5\hat{i}+2\hat{j}-2\hat{k}$, what is $\Vec{u}\cdot \Vec{v}$?
\end{problem}

\begin{problem}
    If $\vec{u}\cdot \vec{v} = 2$, $|\vec{u}|=8$, and $|\vec{v}|=\frac{1}{2}$, what is the angle between $\Vec{u}$ and $\vec{v}$? 
\end{problem}

\begin{problem}
    Are $\langle 1,1,1 \rangle$ and $\langle -2,3,-1\rangle$ orthogonal?
\end{problem}

\begin{problem}
    Find $\text{proj}_ab$ if $a = \langle 1,0,2 \rangle$ and $b = \langle 0,0,3 \rangle$.
\end{problem}

\newpage

\section{Answers}

\begin{enumerate}
    \item $-10$
    \item $-14$
    \item $2 = (8)(\frac{1}{2})\cos(\theta)$ so $\cos(\theta) = \frac{1}{2} \implies \theta = \frac{\pi}{3}$.
    \item Yes since $\langle 1,1,1 \rangle \cdot \langle -2,3,-1\rangle = 0$
    \item $\text{proj}_ab = \langle \frac{6}{5}, 0, \frac{12}{5}\rangle $
\end{enumerate}
