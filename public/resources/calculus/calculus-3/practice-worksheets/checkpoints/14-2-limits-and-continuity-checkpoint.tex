\vs\vs

\begin{problem}
What is $z=f(x,y)$? How many independent and dependent variables does it have?
\end{problem}

\begin{problem}
Why is it harder to find determine whether a limit exists on a surface than on a curve?
\end{problem}

\begin{problem}
Find $\lim_{(x,y)\to (1,1)} e^xy+2xy-y^2$
\end{problem}


\begin{problem}
Is the following statement valid or invalid?

``The $\lim_{(x,y)\to (a,b)}f(x,y)$ exists and is equal to $c$ since along the path $x = a$, $f(x,y)$ approaches $c$, and along the path $y=b$, $f(x,y)$ approaches $c$"
\end{problem}

\begin{problem}
Determine if $\lim_{(x,y)\to (0,0)}\frac{xy^2}{x^2+y^4}$ exists or does not exist.
\end{problem}


\newpage

\section{Answers}

\begin{enumerate}
    \item A surface with one dependent variable $z$ and two independent variables $x,y$.
    \item There are infinitely many paths along my independent variables to get to the desired point $(x,y)$ whereas for a curve there are just two (left and right).
    \item $e+1$
    \item Invalid, to show a limit does exists we need to prove along \emph{all} paths, $f(x,y)$ approaches $c$.
    \item Does not exist.
\end{enumerate}
