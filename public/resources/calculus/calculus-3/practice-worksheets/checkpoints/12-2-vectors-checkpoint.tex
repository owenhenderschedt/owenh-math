\section{Problems}

\begin{problem}
What two things make up a vector?
\end{problem}

\begin{problem}
What dimension does the following vectors live in and what is their sum?  $ \vec{v}= \langle 2,4,9,-1 \rangle$ and $\vec{u} = \langle 0,-3,-3,2\rangle$
\end{problem}

\begin{problem}
Find the magnitude of $\langle 1,-1,9 \rangle$ and find the magnitude of $-2\hat{i}+5\hat{j}-2\hat{k}$.
\end{problem}

\begin{problem}
What is the unit vector going in the same direction as $2\hat{i}-\hat{j}+3\hat{k}$?
\end{problem}

\begin{problem}
What does it mean for two vectors to be parallel?
\end{problem}

\newpage

\section{Answers}

\begin{enumerate}
    \item a magnitude and a direction.
    \item $4$-dimensions $\vec{v}+\vec{u} = \langle 2,1,6,1\rangle$
    \item $|\langle 1,-1,9 \rangle|=\sqrt{83}$ and $|-2\hat{i}+5\hat{j}-2\hat{k}| = \sqrt{33}$
    \item $\frac{2}{\sqrt{14}}\hat{i}-\frac{1}{\sqrt{14}}\hat{j}+\frac{3}{\sqrt{14}}\hat{k}$
    \item They are scalar multiples. That is $\vec{u} = t \vec{v}$ where $t$ is some real number.
\end{enumerate}
