\vs\vs

\begin{problem}
What is a level curve to a surface $f(x,y)$?
\end{problem}

\begin{problem}
The gradient of a $k$-dimensional surface lives in what dimension? What does the gradient vector represent with respect to a level curve of the surface?
\end{problem}

\begin{problem}
If a plane needs a point and a normal? What will we use for the normal of a \textbf{tangent} plane?
\end{problem}


\begin{problem}
If $2(x-1)+3(y-2)+4(z-3)=0$ is a tangent plane to some surface $f(x,y)$, find the corresponding normal line at the same point.  
\end{problem}

\begin{problem}
If $\nabla f(x,y,z) = \langle xy, x+y+z,x-y-z\rangle$ find the equation of the tangent plane at $(1,2,3)$.
\end{problem}




\newpage

\section{Answers}

\begin{enumerate}
    \item It is setting our surface equal to a constant $f(x,y) = c$.
    \item $k-1$. The gradient is the normal vector to the level curve.
    \item We use the gradient of a surface one dimension higher by thinking about our surface as a specific level surface of something larger.
    \item $\frac{x-1}{2}= \frac{y-2}{3}= \frac{z-3}{4}$.
    \item $2(x-1)+6(y-2)-4(z-3)=0$
\end{enumerate}
