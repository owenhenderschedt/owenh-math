\vs\vs

\begin{problem}
Why is the \emph{derivative} ambiguous in 3-dimensions?
\end{problem}

\begin{problem}
What else do we need in order for a derivative to no longer be ambiguous? What are the most natural choices for these?
\end{problem}

\begin{problem}
Find $f_x$ and $f_y$ where $f(x,y) = 3xy+x^3-7y^2$.
\end{problem}


\begin{problem}
What is $f_{xxxx...x}$ if $f(x,y) = ye^{x}$
\end{problem}

\begin{problem}
Find $\frac{\partial f}{\partial y}$ where $f(x,y) = \ln(xy+x)$.
\end{problem}


\newpage

\section{Answers}

\begin{enumerate}
    \item There are infinitely many lines tangent to a point on a surface.
    \item We need to specify a direction. $\hat{i}$ and $\hat{j}$ are the most natural choices ($x$-direction and $y$-direction).
    \item $f_x = 3y+3x^2$ and $f_y = 3x - 14y$.
    \item $ye^x$
    \item $\frac{x}{xy+x}$
\end{enumerate}
