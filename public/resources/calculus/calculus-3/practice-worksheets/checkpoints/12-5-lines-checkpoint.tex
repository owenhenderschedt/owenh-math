\begin{problem}
What two things do you need for a line in $3$D?  
\end{problem}

\begin{problem}
If I gave you a point on line $L_1$ and the equation of a parallel line $L_2$, how would you find the equation of $L_1$?
\end{problem}

\begin{problem}
Rewrite the equation of the following line in symmetric form. \[x(t) = 2t-5, y(t) = -t+7, z(t) = 3t-4\]
\end{problem}

\begin{problem}
    Rewrite the equation of the following line in parametric form. \[\frac{x+3}{-2} = \frac{y}{11} = z-4\]
\end{problem}

\begin{problem}
Are the following lines intersecting, parallel, or skew?
\[L_1: x(t) = 2t+5, y(t) = t-3, z(t) = 2t\] and \[L_2: \frac{x+1}{4}= \frac{y-4}{2} = \frac{z+2}{4}\]
\end{problem}
