\sectionbar{A. Double Integrals}

\begin{parts}

\item \textbf{Separable double integral.}
Evaluate
\[
\iint_R y\sqrt{9-y^2}\sin(2x)\,dA,
\]
where
\[
R=\left\{(x,y):0\leq x\leq\frac{\pi}{4},\ 0\leq y\leq3\right\}.
\]

\item \textbf{Iterated integral.}
Evaluate
\[
\int_0^1\int_1^3\frac{1}{(2x+y)^2}\,dy\,dx.
\]

\item \textbf{Substitution inside a double integral.}
Let
\[
R=\{(x,y):0\leq x\leq1,\ 0\leq y\leq3\}.
\]
Evaluate
\[
\iint_R xe^{xy}\,dA.
\]

\end{parts}

\sectionbar{B. Lagrange Multipliers}

\begin{parts}

\item \textbf{Positive variables with a product constraint.}
Use Lagrange multipliers to find the positive values of $x,y,z$
that minimize
\[
x+2y+3z
\]
subject to
\[
xyz=36.
\]

\item \textbf{Optimization with a plane constraint.}
Use Lagrange multipliers to find the maximum value of
\[
f(x,y,z)=xy+xz+yz
\]
subject to
\[
x+y+z=12.
\]

\item \textbf{Linear function on a sphere.}
Find the point on
\[
x^2+y^2+z^2=14
\]
where
\[
f(x,y,z)=2x-y+3z
\]
attains its minimum value.

\end{parts}

\sectionbar{C. Absolute Extrema on Closed Regions}

\begin{parts}

\item \textbf{Closed triangular region.}
Find the absolute maximum and absolute minimum values of
\[
f(x,y)=x^2+y^2-4x-2y
\]
on the triangular region bounded by
\[
x=0,\qquad y=0,\qquad x+y=4.
\]
Give the point where each value occurs.

\item \textbf{Closed elliptical region.}
Find the absolute maximum value of
\[
f(x,y)=4+2xy
\]
on
\[
\frac{x^2}{9}+\frac{y^2}{4}\leq1.
\]
Give all points where the maximum occurs.

\item \textbf{Closed rectangular region.}
Find the absolute maximum and absolute minimum values of
\[
f(x,y)=x^2+y^2-4x+2y
\]
on
\[
0\leq x\leq3,\qquad -2\leq y\leq2.
\]
Give the point where each value occurs.

\end{parts}
