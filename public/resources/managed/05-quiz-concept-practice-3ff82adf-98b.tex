\sectionbar{A. Level Curves and Level Surfaces}

\begin{parts}

\item Let
\[
f(x,y,z)
=
\int_0^x e^t\,dt
+
\int_1^y\frac{dt}{t}
+
\sqrt{z-1}.
\]
Find the domain of $f$ and the level surface through
$(\ln 2,e,5)$.

\item For
\[
f(x,y)=
\begin{cases}
\dfrac{x^2+y^2}{2x+y},&2x+y\neq0,\\[0.2em]
0,&2x+y=0,
\end{cases}
\]
sketch and label the level curves of heights $-2$ and $2$.

\item Let
\[
f(x,y,z)
=
\frac{2x-y+z+1}{x+3y-2z+4}.
\]
Find an equation of the level surface through $(1,0,2)$.

\end{parts}

\sectionbar{B. Gradient and Directional Change}

\begin{parts}

\item Find the direction of steepest descent at $(1,0)$ for
\[
f(x,y)=e^y(x^2+xy)+(x-y)^3.
\]

\item For
\[
f(x,y)=x^2+xy+2y^2,
\]
find the unique vector $\mathbf v=\langle a,b\rangle$ at $(1,-1)$
such that $f$ has zero change in the direction of $\mathbf v$,
\[
|\mathbf v|=2\sqrt{10},
\qquad
a>0.
\]

\end{parts}

\sectionbar{C. Partial and Implicit Differentiation}

\begin{parts}

\item For
\[
f(x,y)=\log_{x+1}(y^2+1),
\]
find $f_x$ and $f_y$.

\item The equation
\[
x^2+2y^2+z^2+xyz=5
\]
defines $z$ implicitly near $(1,1,1)$. Find
\[
\frac{\partial z}{\partial y}
\]
at $(1,1,1)$.

\end{parts}

\sectionbar{D. Multivariable Chain Rule}

\begin{parts}

\item Suppose
\[
F(x,y)=H\bigl(p(x,y),q(x,y)\bigr).
\]
At $(1,-1)$,
\[
p=-1,\quad q=2,\quad
p_x=2,\quad p_y=-3,\quad
q_x=1,\quad q_y=4,
\]
and
\[
H_p(-1,2)=3,
\qquad
H_q(-1,2)=2.
\]
Find $F_x(1,-1)$ and $F_y(1,-1)$.

\item Let
\[
g(s,t)=f(s+t^2,\;st,\;s^2-t).
\]
Given
\[
\nabla f(5,2,-1)=\langle3,-2,4\rangle,
\]
find $g_t(1,2)$.

\item Let
\[
f(x,y,z)=xyz+e^{x^2+y^2}.
\]
Using
\[
x=r\cos\theta,
\qquad
y=r\sin\theta,
\]
express $f_r$, $f_\theta$, and $f_z$ in terms of
$r,\theta,z$.

\end{parts}
